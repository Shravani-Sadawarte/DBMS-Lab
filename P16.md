CREATE TABLE emp2\_audit (

&#x20;   empno NUMBER,

&#x20;   old\_name VARCHAR2(20),

&#x20;   old\_salary NUMBER,

&#x20;   old\_category NUMBER,

&#x20;   new\_salary NUMBER,

&#x20;   modification\_date DATE

);





CREATE OR REPLACE TRIGGER salary\_audit

AFTER UPDATE OF salary ON emp2

FOR EACH ROW

BEGIN

&#x20;   INSERT INTO emp2\_audit

&#x20;   (empno, old\_name, old\_salary, old\_category, new\_salary, modification\_date)

&#x20;   VALUES

&#x20;   (:OLD.empno, :OLD.name, :OLD.salary, :OLD.catagory,

&#x20;    :NEW.salary, SYSDATE);



&#x20;   DBMS\_OUTPUT.PUT\_LINE('Updated Employee: ' || :OLD.empno ||

&#x20;                        ' Old Salary: ' || :OLD.salary ||

&#x20;                        ' New Salary: ' || :NEW.salary);

END;

/





UPDATE emp2

SET salary = 25000

WHERE empno = 1;



COMMIT;
