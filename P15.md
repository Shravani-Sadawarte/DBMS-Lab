CREATE OR REPLACE PROCEDURE min\_max\_salary

AS

&#x20;   min\_sal NUMBER;

&#x20;   max\_sal NUMBER;

&#x20;   low\_sal NUMBER;

&#x20;   upper\_sal NUMBER;

BEGIN

&#x20;   SELECT MIN(salary), MAX(salary)

&#x20;   INTO min\_sal, max\_sal

&#x20;   FROM emp2;



&#x20;   low\_sal := max\_sal / 10;

&#x20;   upper\_sal := min\_sal \* 10;



&#x20;   DBMS\_OUTPUT.PUT\_LINE('Lower Salary: ' || low\_sal);

&#x20;   DBMS\_OUTPUT.PUT\_LINE('Upper Salary: ' || upper\_sal);

END;

/



SET SERVEROUTPUT ON;



BEGIN

&#x20;   min\_max\_salary;

END;

/



=> OUTPUT:

Lower Salary: 3000

Upper Salary: 100000
