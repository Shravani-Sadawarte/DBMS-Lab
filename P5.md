***Que. Swap Numbers without using 3rd variables***



SET SERVEROUTPUT ON



create or replace procedure swap\_num(

&nbsp;   a IN OUT NUMBER,

&nbsp;   b IN OUT NUMBER

)

AS

BEGIN

&nbsp;   DBMS\_OUTPUT.PUT\_LINE('Before Swap: a = ' || a || ', b = ' || b);



&nbsp;   a := a + b;

&nbsp;   b := a - b;

&nbsp;   a := a - b;



&nbsp;   DBMS\_OUTPUT.PUT\_LINE('After Swap:  a = ' || a || ', b = ' || b);

END;

/



**=>Result**



variable c1 number;

variable c2 number;

execute :c1 := 10

execute :c2 := 20

execute swap\_num(:c1,:c2);
