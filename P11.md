=>**Statement level trigger :**



CREATE OR REPLACE TRIGGER st\_tri

BEFORE UPDATE

ON st\_info

BEGIN

&#x20;   DBMS\_OUTPUT.PUT\_LINE('This is the statement level trigger fired');

END;

/





=>**Row level Trigger :**



CREATE OR REPLACE TRIGGER row\_tri

BEFORE UPDATE

ON st\_info

FOR EACH ROW

BEGIN

&#x20;   DBMS\_OUTPUT.PUT\_LINE('Row Level Trigger Fired for PRN: ' || :OLD.prn);

END;

/
